## R3A 其餘元件代碼（自己判斷）

```sql
WITH P AS (
  SELECT PNLNAME, FIELDNUM, FIELDTYPE, RECNAME, FIELDNAME, SUBPNLNAME, GOTOPNLGRPNAME,
         FIELDLEFT, FIELDTOP, FIELDRIGHT, FIELDBOTTOM
    FROM PSPNLFIELD
   WHERE PNLNAME LIKE 'TW\_%' ESCAPE '\'
     AND FIELDTYPE IN (10, 12, 15, 16, 18, 21, 23, 24, 25, 27)
),
F AS (
  SELECT P.FIELDTYPE,
         CASE WHEN TRIM(P.RECNAME) IS NOT NULL THEN 1 ELSE 0 END AS HASREC,
         CASE WHEN TRIM(P.SUBPNLNAME) IS NOT NULL THEN 1 ELSE 0 END AS HASSUB,
         CASE WHEN TRIM(P.GOTOPNLGRPNAME) IS NOT NULL THEN 1 ELSE 0 END AS HASGOTO,
         CASE WHEN D.FIELDTYPE = 1 THEN 1 ELSE 0 END AS ISLONG,
         CASE WHEN EXISTS (SELECT 1 FROM PSPNLFIELD X
                            WHERE X.PNLNAME = P.PNLNAME AND X.FIELDNUM <> P.FIELDNUM
                              AND X.FIELDLEFT >= P.FIELDLEFT AND X.FIELDRIGHT <= P.FIELDRIGHT
                              AND X.FIELDTOP >= P.FIELDTOP AND X.FIELDBOTTOM <= P.FIELDBOTTOM)
              THEN 1 ELSE 0 END AS HASIN
    FROM P
    LEFT JOIN PSDBFIELD D ON D.FIELDNAME = P.FIELDNAME AND TRIM(P.FIELDNAME) IS NOT NULL
),
T AS (
  SELECT FIELDTYPE, COUNT(*) AS CNT,
         ROUND(100 * AVG(HASREC)) AS REC_P, ROUND(100 * AVG(HASSUB)) AS SUB_P,
         ROUND(100 * AVG(HASGOTO)) AS GOTO_P, ROUND(100 * AVG(ISLONG)) AS LONG_P,
         ROUND(100 * AVG(HASIN)) AS IN_P
    FROM F
   GROUP BY FIELDTYPE
)
SELECT FIELDTYPE AS CODE,
       CASE FIELDTYPE WHEN 12 THEN 'BUTTON_PEOPLECODE' WHEN 16 THEN 'LINK_TO_COMPONENT'
                      WHEN 18 THEN 'SECONDARY_PAGE' WHEN 21 THEN 'BUTTON_TO_SECONDARY_PAGE'
                      WHEN 25 THEN 'HTML_AREA' WHEN 27 THEN 'SCROLL_AREA' ELSE 'OTHER' END AS GUESS,
       CNT,
       CASE WHEN FIELDTYPE = 12 AND REC_P >= 90 AND SUB_P < 10 AND GOTO_P < 10 THEN 'OK'
            WHEN FIELDTYPE = 16 AND GOTO_P >= 70 THEN 'OK'
            WHEN FIELDTYPE = 18 AND SUB_P >= 90 AND REC_P < 10 THEN 'OK'
            WHEN FIELDTYPE = 21 AND SUB_P >= 90 AND REC_P >= 50 THEN 'OK'
            WHEN FIELDTYPE = 25 AND LONG_P >= 50 THEN 'OK'
            WHEN FIELDTYPE = 27 AND IN_P >= 80 THEN 'OK'
            WHEN FIELDTYPE IN (12, 16, 18, 21, 25, 27) THEN 'CHECK'
            WHEN GOTO_P >= 70 THEN 'LINK'
            WHEN SUB_P >= 90 THEN 'SUBPAGE_LIKE'
            WHEN IN_P >= 80 THEN 'CONTAINER'
            WHEN IN_P < 20 THEN 'LINE_OR_SINGLE'
            ELSE 'MIXED' END AS RESULT
  FROM T
 ORDER BY FIELDTYPE;
```

判斷依據：12＝綁 Record 欄位、沒有 SUBPNLNAME 也沒有 GOTO 目標（按下去跑 PeopleCode 的按鈕）；16＝有 GOTOPNLGRPNAME（轉到別的 Component）；
18＝有 SUBPNLNAME、沒綁 Record（Secondary Page）；21＝有 SUBPNLNAME、綁 Record（開 secondary page 的按鈕）；25＝綁長字元欄位（HTML Area）；
27＝框裡有其他元件（Scroll Area）。其他代碼只描述特徵：LINK（有 GOTO 目標）、SUBPAGE_LIKE、CONTAINER（框裡有元件）、LINE_OR_SINGLE（框裡沒有元件，像分隔線）、MIXED。

怎麼看：12、16、18、21、25、27 的 RESULT 是 OK＝推測成立；其他代碼照抄 RESULT 的字。回報「全 OK，10=___ 15=___ 23=___ 24=___」或列出 CHECK 的代碼。

## R3B 相關顯示與唯讀位元（自己算）

```sql
WITH P AS (
  SELECT PNLNAME, FIELDNUM, FIELDTYPE, RECNAME, FIELDNAME, FIELDUSE, ASSOCFIELDNUM
    FROM PSPNLFIELD
   WHERE PNLNAME LIKE 'TW\_%' ESCAPE '\'
),
REL AS (
  SELECT * FROM P WHERE ASSOCFIELDNUM > 0
),
CTL AS (
  SELECT DISTINCT C.PNLNAME, C.FIELDNUM, C.FIELDUSE, C.RECNAME, C.FIELDNAME
    FROM P C JOIN REL R ON R.PNLNAME = C.PNLNAME AND R.ASSOCFIELDNUM = C.FIELDNUM
),
CTLP AS (
  SELECT CASE WHEN TRIM(F.EDITTABLE) IS NOT NULL THEN 1 ELSE 0 END AS HASPROMPT
    FROM CTL C
    LEFT JOIN PSRECFIELDDB F ON F.RECNAME = C.RECNAME AND F.FIELDNAME = C.FIELDNAME
),
OTH AS (
  SELECT P.FIELDUSE FROM P
   WHERE P.FIELDTYPE = 4 AND NVL(P.ASSOCFIELDNUM, 0) = 0
     AND NOT EXISTS (SELECT 1 FROM CTL C WHERE C.PNLNAME = P.PNLNAME AND C.FIELDNUM = P.FIELDNUM)
),
X AS (
  SELECT 'REL' AS GRP, FIELDUSE FROM REL UNION ALL
  SELECT 'CTL', FIELDUSE FROM CTL UNION ALL
  SELECT 'OTH', FIELDUSE FROM OTH
),
B AS (
  SELECT POWER(2, LEVEL - 1) AS BIT FROM DUAL CONNECT BY LEVEL <= 32
),
BITS AS (
  SELECT B.BIT,
         ROUND(100 * AVG(CASE WHEN X.GRP = 'REL' THEN SIGN(BITAND(X.FIELDUSE, B.BIT)) END)) AS IN_REL,
         ROUND(100 * AVG(CASE WHEN X.GRP = 'CTL' THEN SIGN(BITAND(X.FIELDUSE, B.BIT)) END)) AS IN_CTL,
         ROUND(100 * AVG(CASE WHEN X.GRP = 'OTH' THEN SIGN(BITAND(X.FIELDUSE, B.BIT)) END)) AS IN_OTH
    FROM B CROSS JOIN X
   GROUP BY B.BIT
),
S AS (
  SELECT (SELECT COUNT(*) FROM REL) AS REL_N,
         (SELECT ROUND(100 * AVG(CASE WHEN FIELDTYPE = 4 THEN 1 ELSE 0 END)) FROM REL) AS REL_EB,
         (SELECT ROUND(100 * AVG(HASPROMPT)) FROM CTLP) AS CTL_PROMPT,
         (SELECT LISTAGG(BIT, '+') WITHIN GROUP (ORDER BY BIT) FROM BITS WHERE IN_REL >= 95 AND IN_OTH <= 50) AS DO_BITS,
         (SELECT LISTAGG(BIT, '+') WITHIN GROUP (ORDER BY BIT) FROM BITS
           WHERE IN_CTL >= 95 AND IN_REL <= 50 AND IN_OTH <= 50) AS DC_BITS
    FROM DUAL
)
SELECT 'REL_N=' || REL_N || ' REL_EB=' || NVL(TO_CHAR(REL_EB), 'NA') || ' CTL_PROMPT=' || NVL(TO_CHAR(CTL_PROMPT), 'NA')
    || ' DO=' || NVL(DO_BITS, 'NONE') || ' DC=' || NVL(DC_BITS, 'NONE')
    || CASE WHEN REL_EB >= 80 AND CTL_PROMPT >= 60 THEN ' ASSOC=RELATED_DISPLAY' ELSE ' ASSOC=CHECK' END AS RESULT
  FROM S;
```

算法：ASSOCFIELDNUM 大於 0 的元件當成「相關顯示」（代碼旁的說明欄），它指到的元件是「顯示控制欄位」（代碼本身）。相關顯示一定唯讀，
所以在相關顯示裡幾乎都有、在一般 Edit Box 裡多半沒有的位元，就是 FIELDUSE 的「唯讀」；顯示控制欄位專有的位元是「Display Control」。

怎麼看：一行字。ASSOC=RELATED_DISPLAY＝ASSOCFIELDNUM 確實是相關顯示（相關顯示 ≥ 80% 是 Edit Box、它指到的代碼欄位 ≥ 60% 有 prompt）；
DO＝唯讀的位元、DC＝Display Control 的位元（NONE＝沒找到）。整行照抄。

## R3C 鍵與必填（參考）

```sql
SELECT 'KEY_REQ=' || NVL(TO_CHAR(ROUND(100 * AVG(CASE WHEN BITAND(USEEDIT, 1) > 0 THEN SIGN(BITAND(USEEDIT, 256)) END))), 'NA')
    || ' NONKEY_REQ=' || NVL(TO_CHAR(ROUND(100 * AVG(CASE WHEN BITAND(USEEDIT, 1) = 0 THEN SIGN(BITAND(USEEDIT, 256)) END))), 'NA')
       AS RESULT
  FROM PSRECFIELDDB
 WHERE RECNAME LIKE 'TW\_%' ESCAPE '\';
```

怎麼看：KEY_REQ＝客製 Record 的鍵欄位帶 256 的比例、NONKEY_REQ＝非鍵欄位帶 256 的比例。兩個數字照抄（256 是不是「必填」的旁證）。
